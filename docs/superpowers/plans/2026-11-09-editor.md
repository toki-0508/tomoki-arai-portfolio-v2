# 編集モード Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 手元の Mac だけで動く「編集モード」を作り、サイトの文章と画像をブラウザ上で書き換えて下書き(`.editor/`)に保存し、Claude が整えて公開できるようにする。

**Architecture:** 直書きの文言を `copy.json`、プロフィール系データを `site.json` に集め、表示箇所を小さな部品(`C` `S` `F` など)で包む。本番ではこれらは値をそのまま返すだけ。`vite --mode edit` のときだけ、Vite プラグインが部品を編集用の実装に差し替え、`src/content/` の読み込みを下書き優先にし、保存窓口 `/__editor/*` を開く。

**Tech Stack:** React 18 / Vite 6 / Node 22(`node --test`)/ ヘッドレス Chrome。依存パッケージは追加しない。

設計書: [docs/superpowers/specs/2026-11-09-editor-design.md](../specs/2026-11-09-editor-design.md)

`scripts/text-snapshot.mjs`:

```js
#!/usr/bin/env node
// ============================================================
// 全ページの表示テキストを1つのファイルに書き出す。
// 文言の移し替えや公開の前後で、意図しない文字の変化がないかを比べるのに使う。
//
//   npm run build
//   npm run snapshot -- output/snapshots/before.txt
//
// dist を vite preview で配信し、ヘッドレスの Chrome で各ページを開いて
// 描画後の DOM を取り出す。3D(WebGL)は無効にして、毎回同じ結果になるようにする
// (3D が読み込まれたときだけ出る文言は、この比較の対象外)。
// ============================================================
import { execFileSync, spawn } from 'node:child_process';
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';

const ROOT = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const CHROME = process.env.CHROME_PATH || '/Applications/Google Chrome.app/Contents/MacOS/Google Chrome';
const PORT = 4179;
const ATTRS = ['aria-label', 'alt', 'placeholder', 'title'];

const ENTITIES = { amp: '&', lt: '<', gt: '>', quot: '"', apos: "'", '#39': "'", nbsp: ' ' };
const decode = (text) =>
  text.replace(/&(#x[0-9a-f]+|#\d+|[a-z]+|#39);/gi, (m, name) => {
    if (name[0] === '#' && name[1]?.toLowerCase() === 'x') return String.fromCodePoint(parseInt(name.slice(2), 16));
    if (name[0] === '#' && name !== '#39') return String.fromCodePoint(Number(name.slice(1)));
    return ENTITIES[name.toLowerCase()] ?? m;
  });

export function extractText(html) {
  const lines = [];
  const title = /<title>([\s\S]*?)<\/title>/i.exec(html);
  if (title) lines.push(`[title] ${decode(title[1]).trim()}`);

  const bodyMatch = /<body[^>]*>([\s\S]*)<\/body>/i.exec(html);
  const body = (bodyMatch ? bodyMatch[1] : html).replace(/<(script|style)\b[\s\S]*?<\/\1>/gi, '');

  for (const [tag] of body.matchAll(/<[a-zA-Z][^>]*>/g)) {
    for (const attr of ATTRS) {
      const found = new RegExp(`\\s${attr}="([^"]*)"`).exec(tag);
      if (found && found[1].trim()) lines.push(`[${attr}] ${decode(found[1])}`);
    }
  }
  for (const line of decode(body.replace(/<[^>]+>/g, '\n')).split('\n')) {
    if (line.trim()) lines.push(line.trim());
  }
  return lines;
}

export function routes(root = ROOT) {
  const works = fs
    .readdirSync(path.join(root, 'src/content/works'))
    .filter((file) => file.endsWith('.md'))
    .map((file) => `/works/${file.slice(0, -3)}`)
    .sort();
  return ['/', '/about', '/works', '/downloads', '/contact', ...works];
}

async function waitForServer(url, tries = 50) {
  for (let i = 0; i < tries; i++) {
    try {
      if ((await fetch(url)).ok) return;
    } catch {
      /* まだ起動していない */
    }
    await new Promise((resolve) => setTimeout(resolve, 200));
  }
  throw new Error(`プレビューサーバーが起動しませんでした: ${url}`);
}

async function main() {
  const out = process.argv[2];
  if (!out) {
    console.error('使い方: npm run snapshot -- <出力ファイル>');
    process.exit(1);
  }
  if (!fs.existsSync(path.join(ROOT, 'dist/index.html'))) {
    console.error('dist がありません。先に npm run build を実行してください。');
    process.exit(1);
  }

  const vite = path.join(ROOT, 'node_modules/.bin/vite');
  const preview = spawn(vite, ['preview', '--host', '127.0.0.1', '--port', String(PORT), '--strictPort'], { cwd: ROOT, stdio: 'ignore' });
  const profile = fs.mkdtempSync(path.join(os.tmpdir(), 'snapshot-chrome-'));
  try {
    const base = `http://127.0.0.1:${PORT}/`;
    await waitForServer(base);
    const sections = [];
    for (const route of routes()) {
      const html = execFileSync(
        CHROME,
        ['--headless=new', '--disable-gpu', '--disable-3d-apis', '--no-first-run', '--no-default-browser-check',
          `--user-data-dir=${profile}`, '--virtual-time-budget=4000', '--dump-dom', `${base}#${route}`],
        { encoding: 'utf8', maxBuffer: 32 * 1024 * 1024, stdio: ['ignore', 'pipe', 'ignore'] },
      );
      sections.push(`### ${route}`, ...extractText(html), '');
      console.log(`  ${route}`);
    }
    fs.mkdirSync(path.dirname(path.resolve(out)), { recursive: true });
    fs.writeFileSync(out, sections.join('\n'));
    console.log(`書き出しました: ${out}`);
  } finally {
    preview.kill();
    fs.rmSync(profile, { recursive: true, force: true });
  }
}

if (import.meta.url === pathToFileURL(process.argv[1] ?? '').href) {
  main().catch((err) => {
    console.error(err.message);
    process.exit(1);
  });
}
```

`src/content/copy.js`:

```js
// ============================================================
// ページや部品に直書きしていた文言は copy.json にまとめている
// (編集モードから書き換えられるようにするため)。
//
// キーは「ページ.場所.項目」。共通部品は common.*、ナビ・フッター・タブのタイトルは layout.*。
// 文言にデータを差し込むときは {name} と書き、t(key, { name }) で渡す。
// 表記ルールは site.js の冒頭を参照。
// ============================================================
import COPY from './copy.json';

export { COPY };

export function t(key, vars) {
  const template = COPY[key];
  if (typeof template !== 'string') throw new Error(`copy.json に「${key}」がありません`);
  if (!vars) return template;
  return template.replace(/\{(\w+)\}/g, (match, name) => (name in vars ? String(vars[name]) : match));
}
```

```js
// ---------- 書き換え用: 本文を行の範囲で分ける ----------
// 編集モードは本文を「リード」「各セクションの見出し」「各セクションの本文」に分けて書き換える。
// 表示用の parseSections / parseLead と同じく、「## 」で始まる行をセクションの区切りとみなす。
export function bodyRegions(body) {
  const lines = body.split('\n');
  const headings = [];
  lines.forEach((line, i) => {
    if (/^##\s+/.test(line)) headings.push(i);
  });
  return {
    lines,
    lead: { start: 0, end: headings[0] ?? lines.length },
    sections: headings.map((heading, n) => ({ heading, start: heading + 1, end: headings[n + 1] ?? lines.length })),
  };
}

function trimBlankLines(lines) {
  let start = 0;
  let end = lines.length;
  while (start < end && !lines[start].trim()) start++;
  while (end > start && !lines[end - 1].trim()) end--;
  return lines.slice(start, end);
}

// 本文を、書き換えの入力欄に出す生の文字列に分ける
export function splitBody(body) {
  const { lines, lead, sections } = bodyRegions(body);
  const text = ({ start, end }) => trimBlankLines(lines.slice(start, end)).join('\n');
  return {
    lead: text(lead),
    sections: sections.map((section) => ({
      title: lines[section.heading].replace(/^##\s+/, '').trim(),
      body: text(section),
    })),
  };
}
```

`src/editor/patch.js`:

```js
// ============================================================
// 編集モードの書き換え(純粋関数)。
// 指定した場所だけを書き換え、ほかの行・コメント・順番は1文字も変えない。
// 保存窓口(Node)とテストの両方から使う。
//
// 指定(target)の形
//   copy.json            { key }
//   site.json            { path: 'MVV.0.text' }
//   frontmatter の項目   { field }
//   リード文             { part: 'lead' }
//   セクション           { section: 番号, part: 'title' | 'body' }
// ============================================================
import { bodyRegions } from '../lib/markdown-parse.js';
import { COPY_FILE, SITE_FILE } from './shared.js';

export class PatchError extends Error {}

const WORK_FILE = /^src\/content\/works\/[a-z0-9][a-z0-9-]*\.md$/;
const DOWNLOAD_FILE = /^src\/content\/downloads\/[a-z0-9][a-z0-9-]*\.md$/;
const IMAGE_VALUE = /^\/images\/projects\/[a-z0-9][a-z0-9-]*\.(png|jpg)$/;
const FRONTMATTER = /^---\r?\n([\s\S]*?)\r?\n---\r?\n?/;

// 書き換えられる frontmatter の項目と、値の種類
export const WORK_FIELDS = {
  name: 'required',
  summary: 'text',
  role: 'text',
  status: 'text',
  outcome: 'text',
  date: 'date',
  tags: 'list',
  steps: 'list',
  url: 'url',
  github: 'url',
  imageAlt: 'text',
  imageCaption: 'text',
  imageFit: 'fit',
};
export const DOWNLOAD_FIELDS = {
  name: 'required',
  date: 'date',
  size: 'text',
  length: 'text',
  platform: 'text',
  note: 'text',
  tags: 'list',
};

export function fileKind(file) {
  if (file === COPY_FILE) return 'copy';
  if (file === SITE_FILE) return 'site';
  if (WORK_FILE.test(file)) return 'work';
  if (DOWNLOAD_FILE.test(file)) return 'download';
  return null;
}

function oneLine(value) {
  const text = String(value ?? '').trim();
  if (/[\r\n]/.test(text)) throw new PatchError('1行で入力してください');
  return text;
}

function requireLine(value) {
  const text = oneLine(value);
  if (!text) throw new PatchError('空にはできません');
  return text;
}

function normalizeField(type, value) {
  if (type === 'list') {
    if (!Array.isArray(value)) throw new PatchError('一覧の形で送ってください');
    return value.map(oneLine).filter(Boolean);
  }
  const text = oneLine(value);
  if (type === 'required' && !text) throw new PatchError('空にはできません');
  if (type === 'date' && !/^\d{4}-(0[1-9]|1[0-2])$/.test(text)) throw new PatchError('日付は 2026-10 のように「年-月」で入力してください');
  if (type === 'url' && text && !/^https?:\/\/\S+$/.test(text)) throw new PatchError('URL は https:// から入力してください');
  if (type === 'fit' && text && text !== 'contain') throw new PatchError('画像の表示は「全体を表示(contain)」か空にしてください');
  return text;
}

// ---------- JSON ----------
function setJsonString(raw, path, value) {
  if (path.some((key) => key === '__proto__' || key === 'constructor' || key === 'prototype')) {
    throw new PatchError('書き換えられない場所です');
  }
  const data = JSON.parse(raw);
  let parent = data;
  for (const key of path.slice(0, -1)) parent = parent?.[key];
  const last = path[path.length - 1];
  if (parent === null || typeof parent !== 'object' || !Object.hasOwn(parent, last) || typeof parent[last] !== 'string') {
    throw new PatchError(`書き換えられない場所です: ${path.join('.')}`);
  }
  parent[last] = value;
  return `${JSON.stringify(data, null, 2)}\n`;
}

// ---------- frontmatter ----------
// 既存のパーサー(parseScalar)で同じ値として読めるよう、紛らわしい値は引用符で囲む
const needsQuotes = (text) =>
  /\s#/.test(text) || /^[\s"'[#]/.test(text) || /\s$/.test(text) || text === 'true' || text === 'false';
const formatScalar = (text) => (needsQuotes(text) ? `"${text}"` : text);

function setFrontmatterField(raw, field, value) {
  const match = FRONTMATTER.exec(raw);
  if (!match) throw new PatchError('frontmatter が見つかりません');
  const open = /^---\r?\n/.exec(raw)[0];
  const close = match[0].slice(open.length + match[1].length);
  const lines = match[1].split('\n');

  const start = lines.findIndex((line) => line.startsWith(`${field}:`));
  let end = start + 1;
  if (start >= 0) while (end < lines.length && /^\s+-\s+/.test(lines[end])) end++;
  const wasBlock = start >= 0 && (end > start + 1 || lines[start].trim() === `${field}:`);

  let next;
  if (Array.isArray(value)) {
    if (!value.length) next = [];
    else if (wasBlock || value.some((item) => /[,[\]]/.test(item))) next = [`${field}:`, ...value.map((item) => `  - ${formatScalar(item)}`)];
    else next = [`${field}: [${value.map(formatScalar).join(', ')}]`];
  } else {
    next = value ? [`${field}: ${formatScalar(value)}`] : [];
  }

  if (start >= 0) lines.splice(start, end - start, ...next);
  else lines.push(...next);
  return `${open}${lines.join('\n')}${close}${raw.slice(match[0].length)}`;
}

// ---------- 本文 ----------
// 入力を行に整える: 末尾の空白と前後の空行を落とし、続く空行は1行にまとめる
function blockLines(value, { allowEmpty }) {
  const out = [];
  for (const line of String(value ?? '').replace(/\r\n?/g, '\n').split('\n').map((l) => l.replace(/\s+$/, ''))) {
    if (/^##\s+/.test(line)) throw new PatchError('本文に「## 」で始まる行は使えません。見出しはページの見出しを直接直してください');
    if (!line && (!out.length || !out[out.length - 1])) continue;
    out.push(line);
  }
  while (out.length && !out[out.length - 1]) out.pop();
  if (!out.length && !allowEmpty) throw new PatchError('本文を空にはできません');
  return out;
}

// 範囲を「空行 + 本文 + 空行」に置き換える(見出しや次の見出しとの間を1行あける)
function replaceRegion(lines, { start, end }, body) {
  const region = body.length ? ['', ...body, ''] : [''];
  return [...lines.slice(0, start), ...region, ...lines.slice(end)].join('\n');
}

function patchBody(raw, target, value) {
  const head = FRONTMATTER.exec(raw)?.[0] ?? '';
  const { lines, lead, sections } = bodyRegions(raw.slice(head.length));
  if (target.part === 'lead') return head + replaceRegion(lines, lead, blockLines(value, { allowEmpty: true }));

  const section = Number.isInteger(target.section) ? sections[target.section] : undefined;
  if (!section) throw new PatchError('セクションが見つかりません');
  if (target.part === 'title') {
    const title = requireLine(value).replace(/^#+\s*/, '');
    if (!title) throw new PatchError('空にはできません');
    lines[section.heading] = `## ${title}`;
    return head + lines.join('\n');
  }
  return head + replaceRegion(lines, section, blockLines(value, { allowEmpty: false }));
}

// ---------- 入口 ----------
export function applyPatch(file, raw, target = {}, value, { allowImage = false } = {}) {
  const kind = fileKind(file);
  if (kind === 'copy') {
    if (typeof target.key !== 'string') throw new PatchError('文言のキーがありません');
    return setJsonString(raw, [target.key], requireLine(value));
  }
  if (kind === 'site') {
    if (typeof target.path !== 'string') throw new PatchError('場所の指定がありません');
    return setJsonString(raw, target.path.split('.'), requireLine(value));
  }
  if (kind === 'work' || kind === 'download') {
    if (typeof target.field === 'string') {
      if (target.field === 'image') {
        if (!allowImage) throw new PatchError('画像は「画像を差し替え」から変更してください');
        if (!IMAGE_VALUE.test(String(value))) throw new PatchError('画像のパスが正しくありません');
        return setFrontmatterField(raw, 'image', value);
      }
      const type = (kind === 'work' ? WORK_FIELDS : DOWNLOAD_FIELDS)[target.field];
      if (!type) throw new PatchError(`「${target.field}」はここでは変更できません`);
      return setFrontmatterField(raw, target.field, normalizeField(type, value));
    }
    if (['lead', 'title', 'body'].includes(target.part)) return patchBody(raw, target, value);
  }
  throw new PatchError('この書き換えには対応していません');
}
```

```js
// 編集モードの下書き置き場(.editor/、Git 管理外)
//
//   .editor/drafts/<本物と同じ相対パス>   下書き(画像は drafts/public/images/projects/)
//   .editor/manifest.json               下書きごとに「作り始めたときの本物のハッシュ」
//   .editor/notes.md                    Claude へのメモ
//
// 本物(src/content/ など)には書き込まない。反映は scripts/editor-apply.mjs が行う。
// ============================================================
import crypto from 'node:crypto';
import fs from 'node:fs';
import path from 'node:path';
import { parseFrontmatter } from '../../src/lib/markdown-parse.js';
import { fileKind } from '../../src/editor/patch.js';

export class EditorError extends Error {
  constructor(status, message) {
    super(message);
    this.status = status;
  }
}

export const MAX_IMAGE_BYTES = 10 * 1024 * 1024;
const IMAGE_DIR = 'public/images/projects';
const IMAGE_NAME = /^[a-z0-9][a-z0-9-]*\.(png|jpg)$/;
const PNG_MAGIC = Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]);

export const sha256 = (data) => crypto.createHash('sha256').update(data).digest('hex');

// 拡張子ではなく、ファイル先頭のバイトで形式を見分ける
export function imageType(buffer) {
  if (buffer.length >= 8 && buffer.subarray(0, 8).equals(PNG_MAGIC)) return 'png';
  if (buffer.length >= 3 && buffer[0] === 0xff && buffer[1] === 0xd8 && buffer[2] === 0xff) return 'jpg';
  return null;
}

const ymd = (date) =>
  `${date.getFullYear()}${String(date.getMonth() + 1).padStart(2, '0')}${String(date.getDate()).padStart(2, '0')}`;

// <id>-<YYYYMMDD>[-n].<ext> だけを、その作品の画像とみなす(sample と sample-model を取り違えない)
const isImageOf = (id, name) => new RegExp(`^${id}-\\d{8}(-\\d+)?\\.(png|jpg)$`).test(name);

export class DraftStore {
  constructor(root) {
    this.root = root;
    this.dir = path.join(root, '.editor');
    this.draftsDir = path.join(this.dir, 'drafts');
    this.manifestPath = path.join(this.dir, 'manifest.json');
    this.notesPath = path.join(this.dir, 'notes.md');
  }

  assertContentFile(rel) {
    const valid =
      typeof rel === 'string' &&
      !rel.includes('\\') &&
      path.posix.normalize(rel) === rel &&
      !rel.split('/').includes('..') &&
      fileKind(rel) !== null;
    if (!valid) throw new EditorError(403, 'このファイルは編集できません');
    if (!fs.existsSync(path.join(this.root, rel))) throw new EditorError(404, 'ファイルが見つかりません');
    return rel;
  }

  draftPath(rel) {
    const full = path.resolve(this.draftsDir, rel);
    if (!full.startsWith(this.draftsDir + path.sep)) throw new EditorError(403, 'このファイルは編集できません');
    return full;
  }

  readManifest() {
    try {
      return JSON.parse(fs.readFileSync(this.manifestPath, 'utf8'));
    } catch {
      return {};
    }
  }

  writeManifest(manifest) {
    fs.mkdirSync(this.dir, { recursive: true });
    fs.writeFileSync(this.manifestPath, `${JSON.stringify(manifest, null, 2)}\n`);
  }

  original(rel) {
    return fs.readFileSync(path.join(this.root, rel), 'utf8');
  }

  hasDraft(rel) {
    try {
      return fs.existsSync(this.draftPath(rel));
    } catch {
      return false;
    }
  }

  read(rel) {
    return this.hasDraft(rel) ? fs.readFileSync(this.draftPath(rel), 'utf8') : this.original(rel);
  }

  write(rel, text) {
    this.assertContentFile(rel);
    const original = this.original(rel);
    const manifest = this.readManifest();
    const target = this.draftPath(rel);
    if (text === original) {
      fs.rmSync(target, { force: true });
      this.removeDraftImages(rel);
      delete manifest[rel];
    } else {
      manifest[rel] ??= { base: sha256(original) };
      fs.mkdirSync(path.dirname(target), { recursive: true });
      fs.writeFileSync(target, text);
    }
    this.writeManifest(manifest);
  }

  imagesDir() {
    return path.join(this.draftsDir, IMAGE_DIR);
  }

  itemId(rel) {
    return path.posix.basename(rel, '.md');
  }

  draftImagesOf(rel) {
    if (!fs.existsSync(this.imagesDir())) return [];
    const id = this.itemId(rel);
    return fs.readdirSync(this.imagesDir()).filter((name) => isImageOf(id, name)).sort();
  }

  removeDraftImages(rel) {
    for (const name of this.draftImagesOf(rel)) fs.rmSync(path.join(this.imagesDir(), name), { force: true });
  }

  saveImage(rel, buffer, { now = new Date() } = {}) {
    this.assertContentFile(rel);
    if (!['work', 'download'].includes(fileKind(rel))) throw new EditorError(403, '画像を差し替えられるのは作品とダウンロードだけです');
    if (buffer.length > MAX_IMAGE_BYTES) throw new EditorError(413, '画像が大きすぎます(10MBまで)');
    const ext = imageType(buffer);
    if (!ext) throw new EditorError(415, 'PNG か JPEG の画像を選んでください');

    this.removeDraftImages(rel);
    const base = `${this.itemId(rel)}-${ymd(now)}`;
    let name = `${base}.${ext}`;
    for (let n = 2; fs.existsSync(path.join(this.root, IMAGE_DIR, name)); n++) name = `${base}-${n}.${ext}`;
    fs.mkdirSync(this.imagesDir(), { recursive: true });
    fs.writeFileSync(path.join(this.imagesDir(), name), buffer);
    return `/images/projects/${name}`;
  }

  draftImagePath(name) {
    if (!IMAGE_NAME.test(name)) return null;
    const full = path.join(this.imagesDir(), name);
    return fs.existsSync(full) ? full : null;
  }

  discard(rel) {
    this.assertContentFile(rel);
    fs.rmSync(this.draftPath(rel), { force: true });
    this.removeDraftImages(rel);
    const manifest = this.readManifest();
    delete manifest[rel];
    this.writeManifest(manifest);
  }

  discardAll() {
    fs.rmSync(this.draftsDir, { recursive: true, force: true });
    this.writeManifest({});
  }

  label(rel) {
    const kind = fileKind(rel);
    if (kind === 'copy') return 'サイトの文言';
    if (kind === 'site') return 'プロフィール・サイトの情報';
    const name = parseFrontmatter(this.read(rel)).data.name || this.itemId(rel);
    return `${kind === 'work' ? '作品' : 'ダウンロード'}「${name}」`;
  }

  list() {
    const manifest = this.readManifest();
    return Object.keys(manifest)
      .filter((rel) => this.hasDraft(rel) && fs.existsSync(path.join(this.root, rel)))
      .sort()
      .map((rel) => ({
        file: rel,
        label: this.label(rel),
        conflict: sha256(this.original(rel)) !== manifest[rel].base,
        images: this.draftImagesOf(rel),
      }));
  }

  getNotes() {
    try {
      return fs.readFileSync(this.notesPath, 'utf8');
    } catch {
      return '';
    }
  }

  setNotes(text) {
    if (text.length > 20000) throw new EditorError(413, 'メモが長すぎます');
    fs.mkdirSync(this.dir, { recursive: true });
    fs.writeFileSync(this.notesPath, text);
  }
}
```

```js
const LOCAL_HOST = /^(127\.0\.0\.1|localhost)(:\d{1,5})?$/;

// このMacの編集画面から来たリクエストだけを受け付ける。
// X-Editor ヘッダーは、ほかのサイトのページからは(事前確認なしに)付けられない。
export function isTrustedRequest(headers = {}) {
  const host = headers.host ?? '';
  if (!LOCAL_HOST.test(host)) return false;
  if (headers['x-editor'] !== '1') return false;
  const { origin } = headers;
  return !origin || origin === `http://${host}`;
}

export function readBody(req, limit) {
  return new Promise((resolve, reject) => {
    const chunks = [];
    let size = 0;
    req.on('data', (chunk) => {
      size += chunk.length;
      if (size > limit) {
        reject(new EditorError(413, '送られたデータが大きすぎます'));
        req.destroy();
        return;
      }
      chunks.push(chunk);
    });
    req.on('end', () => resolve(Buffer.concat(chunks)));
    req.on('error', reject);
  });
}

export async function readJson(req, limit = 1024 * 1024) {
  const body = await readBody(req, limit);
  try {
    return JSON.parse(body.toString('utf8'));
  } catch {
    throw new EditorError(400, '送られたデータを読めませんでした');
  }
}

export function sendJson(res, status, data) {
  res.statusCode = status;
  res.setHeader('Content-Type', 'application/json; charset=utf-8');
  res.setHeader('Cache-Control', 'no-store');
  res.end(JSON.stringify(data));
}
```

```js
// ============================================================
// 編集モード(npm run edit)の Vite プラグイン。dev サーバーでだけ動く。
//
//   1. src/editor/index.jsx(本番用の部品)を src/editor/live/index.jsx に差し替える
//   2. src/content/ のファイルを読み込むとき、下書きがあれば下書きを返す
//   3. /images/projects/ へのリクエストは、下書きの画像があればそれを返す
//   4. 保存窓口 /__editor/* を開く(このMacから、X-Editor ヘッダー付きのときだけ)
// ============================================================
import fs from 'node:fs';
import path from 'node:path';
import { applyPatch, PatchError } from '../../src/editor/patch.js';
import { DraftStore, EditorError, MAX_IMAGE_BYTES } from './drafts.mjs';
import { isTrustedRequest, readBody, readJson, sendJson } from './http.mjs';

export function editorPlugin() {
  let root;
  let store;
  let server = null;

  // 下書きが変わったら、元のファイルを読み込んでいるモジュールを読み込み直す(ページは即座に更新される)
  const reload = (rel) => {
    if (!server) return;
    for (const mod of server.moduleGraph.getModulesByFile(path.join(root, rel)) ?? []) server.reloadModule(mod);
  };

  async function handle(req, res, url) {
    const route = `${req.method} ${url.pathname}`;
    if (route === 'GET /__editor/state') {
      return sendJson(res, 200, { drafts: store.list(), notes: store.getNotes() });
    }
    if (route === 'POST /__editor/patch') {
      const { file, target, value } = await readJson(req);
      store.assertContentFile(file);
      store.write(file, applyPatch(file, store.read(file), target, value));
      reload(file);
      return sendJson(res, 200, { ok: true });
    }
    if (route === 'POST /__editor/image') {
      const file = store.assertContentFile(url.searchParams.get('file'));
      const image = store.saveImage(file, await readBody(req, MAX_IMAGE_BYTES + 1));
      store.write(file, applyPatch(file, store.read(file), { field: 'image' }, image, { allowImage: true }));
      reload(file);
      return sendJson(res, 200, { ok: true, image });
    }
    if (route === 'POST /__editor/discard') {
      const { file } = await readJson(req);
      const files = file ? [store.assertContentFile(file)] : store.list().map((draft) => draft.file);
      if (file) store.discard(file);
      else store.discardAll();
      files.forEach(reload);
      return sendJson(res, 200, { ok: true });
    }
    if (route === 'PUT /__editor/notes') {
      const { text } = await readJson(req);
      store.setNotes(String(text ?? ''));
      return sendJson(res, 200, { ok: true });
    }
    throw new EditorError(404, '窓口が見つかりません');
  }

  function serveDraftImage(pathname, res) {
    let name;
    try {
      name = decodeURIComponent(pathname.slice('/images/projects/'.length));
    } catch {
      return false;
    }
    const draft = store.draftImagePath(name);
    if (!draft) return false;
    res.setHeader('Content-Type', draft.endsWith('.png') ? 'image/png' : 'image/jpeg');
    res.setHeader('Cache-Control', 'no-store');
    fs.createReadStream(draft).pipe(res);
    return true;
  }

  return {
    name: 'portfolio-editor',
    apply: 'serve',
    enforce: 'pre',

    configResolved(config) {
      root = config.root;
      store = new DraftStore(root);
    },

    async resolveId(source, importer, options) {
      if (!importer || !/editor\/index\.jsx$/.test(source)) return null;
      if (importer.includes(`${path.sep}src${path.sep}editor${path.sep}live${path.sep}`)) return null;
      const resolved = await this.resolve(source, importer, { ...options, skipSelf: true });
      return resolved?.id === path.join(root, 'src/editor/index.jsx') ? path.join(root, 'src/editor/live/index.jsx') : null;
    },

    load(id) {
      const [file, query = ''] = id.split('?');
      if (!file.startsWith(path.join(root, 'src', 'content') + path.sep)) return null;
      const rel = path.relative(root, file).split(path.sep).join('/');
      if (!store.hasDraft(rel)) return null;
      const text = store.read(rel);
      // ?raw は文字列として、.json はそのまま返す(JSON への変換は Vite がこの後に行う)
      return query.split('&').includes('raw') ? `export default ${JSON.stringify(text)}` : text;
    },

    configureServer(devServer) {
      server = devServer;
      // editor:apply などで下書きが外から変わったときも、ページに反映する
      server.watcher.add(store.draftsDir);
      const onDraftChange = (file) => {
        const rel = path.relative(store.draftsDir, file).split(path.sep).join('/');
        if (rel.startsWith('src/content/')) reload(rel);
      };
      for (const event of ['add', 'change', 'unlink']) server.watcher.on(event, onDraftChange);

      server.middlewares.use(async (req, res, next) => {
        const url = new URL(req.url, 'http://localhost');
        if (req.method === 'GET' && url.pathname.startsWith('/images/projects/')) {
          if (!serveDraftImage(url.pathname, res)) next();
          return;
        }
        if (!url.pathname.startsWith('/__editor/')) {
          next();
          return;
        }
        try {
          if (!isTrustedRequest(req.headers)) throw new EditorError(403, 'このリクエストは受け付けられません');
          await handle(req, res, url);
        } catch (err) {
          const status = err instanceof EditorError ? err.status : err instanceof PatchError ? 400 : 500;
          sendJson(res, status, { error: err.message });
        }
      });
    },
  };
}
```

```js
export const MARKERS = ['/__editor/', 'data-portfolio-editable', 'portfolio-editor-bar'];

export function findMarkers(dir) {
  const hits = [];
  const walk = (current) => {
    for (const entry of fs.readdirSync(current, { withFileTypes: true })) {
      const full = path.join(current, entry.name);
      if (entry.isDirectory()) {
        walk(full);
```

```js
import { useSyncExternalStore } from 'react';

const API = '/__editor';
let state = { mode: 'edit', drafts: [], notes: '', error: '', panel: null, focusField: null };
const listeners = new Set();

function set(patch) {
  state = { ...state, ...patch };
  for (const listener of listeners) listener();
}

export const getState = () => state;
const subscribe = (listener) => {
  listeners.add(listener);
  return () => listeners.delete(listener);
};
export const useEditorState = () => useSyncExternalStore(subscribe, getState);

async function request(method, url, body, type = 'application/json') {
  const init = { method, headers: { 'X-Editor': '1' } };
  if (body !== undefined) {
    init.headers['Content-Type'] = type;
    init.body = type === 'application/json' ? JSON.stringify(body) : body;
  }
  const res = await fetch(`${API}${url}`, init);
  const data = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error(data.error || `保存できませんでした(${res.status})`);
  return data;
}

    set({ error: '' });
    return result ?? true;
  } catch (err) {
    set({ error: err.message });
    return false;
  }
}

export const refresh = () =>
  run(async () => {
    const data = await request('GET', '/state');
```

```jsx
// ============================================================
// 編集できる文章。
//   kind="line"   … その場で1行を書き換える(Enter で確定、Esc で取り消し、外をクリックで確定)
//   kind="blocks" … 本文を入力欄に切り替える(空行で段落、「- 」で箇条書き。Cmd+Enter か外をクリックで確定)
//   kind="list"   … 1行に1項目(steps など)
// クリックは useClickCapture がまとめて受け取る(リンクやボタンは動かさない)。
// ============================================================
import { createElement, Fragment, useEffect, useLayoutEffect, useRef, useState } from 'react';
import { getState, savePatch, showError } from './store.js';

// 編集できる要素 → 編集を始める関数
const openers = new WeakMap();

// 文言の中の {name} の差し込みを並べた文字列(変わったかどうかの比較用)
const placeholders = (text) => (String(text).match(/\{\w+\}/g) ?? []).sort().join(' ');

export function useClickCapture() {
  useEffect(() => {
    const onClick = (event) => {
      if (getState().mode !== 'edit') return;
      if (!(event.target instanceof Element) || event.target.closest('[data-editor-ui]')) return;
      // ページ側のリンク・ボタン・フォームは動かさない(閲覧に切り替えると動く)
      event.preventDefault();
      event.stopPropagation();
      const el = event.target.closest('[data-portfolio-editable]');
      if (el && !el.classList.contains('is-editing')) openers.get(el)?.();
    };
    document.addEventListener('click', onClick, true);
    return () => document.removeEventListener('click', onClick, true);
  }, []);
}

function LineEditor({ raw, hint, onDone }) {
  const ref = useRef(null);
  const done = useRef(false);

  useLayoutEffect(() => {
    const el = ref.current;
    el.focus();
    const range = document.createRange();
    range.selectNodeContents(el);
    range.collapse(false);
    window.getSelection().removeAllRanges();
    window.getSelection().addRange(range);
  }, []);

  const finish = (save) => {
    if (done.current) return;
    done.current = true;
    onDone(save ? ref.current.textContent.replace(/\s*\n\s*/g, ' ').trim() : null);
  };

  return (
    <Fragment>
      <span
        ref={ref}
        className="ped is-editing"
        data-portfolio-editable="line"
        contentEditable="plaintext-only"
        suppressContentEditableWarning
        spellCheck={false}
        onKeyDown={(event) => {
          if (event.key === 'Enter') {
            // 日本語入力の変換を確定する Enter では確定しない
            if (event.nativeEvent.isComposing || event.keyCode === 229) return;
            event.preventDefault();
            finish(true);
          } else if (event.key === 'Escape') {
            event.preventDefault();
            finish(false);
          }
        }}
        onBlur={() => finish(true)}
      >
        {raw}
      </span>
      {hint && <span className="ped-hint" data-editor-ui>{hint}</span>}
    </Fragment>
  );
}

function BlockEditor({ raw, kind, onDone }) {
  const ref = useRef(null);
  const done = useRef(false);
  const fit = () => {
    const el = ref.current;
    el.style.height = 'auto';
    el.style.height = `${el.scrollHeight + 2}px`;
  };

  useLayoutEffect(() => {
    fit();
    ref.current.focus();
  }, []);

  const finish = (save) => {
    if (done.current) return;
    done.current = true;
    if (!save) return onDone(null);
    const text = ref.current.value;
    onDone(kind === 'list' ? text.split('\n').map((line) => line.replace(/^\s*-\s+/, '').trim()).filter(Boolean) : text);
  };

  return (
    <Fragment>
      <textarea
        ref={ref}
        className="ped is-editing ped-textarea"
        data-portfolio-editable={kind}
        defaultValue={kind === 'list' ? raw.join('\n') : raw}
        spellCheck={false}
        onInput={fit}
        onKeyDown={(event) => {
          if (event.key === 'Enter' && (event.metaKey || event.ctrlKey)) {
            event.preventDefault();
            finish(true);
          } else if (event.key === 'Escape') {
            event.preventDefault();
            finish(false);
          }
        }}
        onBlur={() => finish(true)}
      />
      <p className="ped-hint" data-editor-ui>
        {kind === 'list' ? '1行に1項目' : '空行で段落、「- 」で始まる行は箇条書き'}。Cmd+Enter で確定、Esc で取り消し
      </p>
    </Fragment>
  );
}

export function Editable({ file, target, raw, kind = 'line', required = true, hint = null, children }) {
  const ref = useRef(null);
  const [editing, setEditing] = useState(false);

  // 描き直しで要素が入れ替わることがあるので、毎回登録し直す
  useLayoutEffect(() => {
    const el = ref.current;
    if (!el) return undefined;
    openers.set(el, () => setEditing(true));
    return () => openers.delete(el);
  });

  const onDone = (value) => {
    setEditing(false);
    if (value === null || JSON.stringify(value) === JSON.stringify(raw)) return;
    if (required && (value === '' || (Array.isArray(value) && !value.length))) {
      showError('空にはできません。元の文章に戻しました');
      return;
    }
    if (kind === 'line' && placeholders(value) !== placeholders(raw)) {
      const ok = window.confirm(
        `自動で入る部分(${placeholders(raw) || 'なし'})が変わっています。\nこのまま保存しますか?`,
      );
      if (!ok) return;
    }
    savePatch(file, target, value);
  };

  if (editing) {
    return kind === 'line'
      ? <LineEditor raw={raw} hint={hint} onDone={onDone} />
      : <BlockEditor raw={raw} kind={kind} onDone={onDone} />;
  }
  return createElement(kind === 'line' ? 'span' : 'div', { ref, className: 'ped', 'data-portfolio-editable': kind }, children);
```

```js
// ============================================================
// 選んだ画像をブラウザの中で描き直してから保存する。
// - 位置情報(GPS)などの埋め込み情報は、描き直すときにすべて消える
// - 長辺が MAX_SIDE を超える写真は縮小する(サイトを重くしない)
// - PNG は PNG のまま(透明を保つ)、それ以外は JPEG にする
// ============================================================
export const MAX_SIDE = 2400;
const JPEG_QUALITY = 0.88;

export function targetSize(width, height, max = MAX_SIDE) {
  const scale = Math.min(1, max / Math.max(width, height));
  return { width: Math.max(1, Math.round(width * scale)), height: Math.max(1, Math.round(height * scale)) };
}

export const outputType = (file) => (file.type === 'image/png' ? 'image/png' : 'image/jpeg');

export async function reencodeImage(file) {
  let bitmap;
  try {
    // 写真の向き(EXIF の回転)は createImageBitmap が反映してから描く
    bitmap = await createImageBitmap(file);
  } catch {
    throw new Error('この画像は読み込めませんでした。JPEG か PNG で書き出してから選んでください(iPhone の HEIC など)。');
  }
  const { width, height } = targetSize(bitmap.width, bitmap.height);
  const canvas = document.createElement('canvas');
  canvas.width = width;
  canvas.height = height;
  const context = canvas.getContext('2d');
  const type = outputType(file);
  if (type === 'image/jpeg') {
    // JPEG は透明を持てないので、透明な部分は白にする
    context.fillStyle = '#ffffff';
    context.fillRect(0, 0, width, height);
  }
  context.drawImage(bitmap, 0, 0, width, height);
  bitmap.close?.();

  const blob = await new Promise((resolve) => canvas.toBlob(resolve, type, type === 'image/jpeg' ? JPEG_QUALITY : undefined));
  if (!blob) throw new Error('画像を保存できませんでした。別の画像で試してください。');
  return blob;
}
```

```jsx
// ============================================================
// 差し替えられる画像。画像の枠(この部品を置いた親要素)の右上に「画像を差し替え」を出し、
// クリックでファイルを選ぶか、枠の上にドラッグ&ドロップして差し替える。
// ============================================================
import { Fragment, useEffect, useRef, useState } from 'react';
import { createPortal } from 'react-dom';
import { reencodeImage } from './reencode.js';
import { getState, openPanel, showError, uploadImage } from './store.js';

export function EditableImage({ item, children }) {
  const buttonRef = useRef(null);
  const inputRef = useRef(null);
  const [busy, setBusy] = useState(false);

  const replace = async (file) => {
    if (!file || busy) return;
    setBusy(true);
    try {
      const image = await uploadImage(item.source, await reencodeImage(file));
      // 作品の画像は説明文(alt)も確かめてもらう
      if (image && item.source.startsWith('src/content/works/')) openPanel('settings', 'imageAlt');
    } catch (err) {
      showError(err.message);
    } finally {
      setBusy(false);
    }
  };
  const replaceRef = useRef(replace);
  replaceRef.current = replace;

  useEffect(() => {
    const host = buttonRef.current?.parentElement;
    if (!host) return undefined;
    const hasFile = (event) => getState().mode === 'edit' && [...(event.dataTransfer?.types ?? [])].includes('Files');
    const over = (event) => {
      if (!hasFile(event)) return;
      event.preventDefault();
      host.classList.add('is-dropping');
    };
    const leave = () => host.classList.remove('is-dropping');
    const drop = (event) => {
      leave();
      if (!hasFile(event)) return;
      event.preventDefault();
      replaceRef.current(event.dataTransfer.files[0]);
    };
    host.classList.add('ped-image-host');
    host.addEventListener('dragover', over);
    host.addEventListener('dragleave', leave);
    host.addEventListener('drop', drop);
    return () => {
      host.classList.remove('ped-image-host', 'is-dropping');
      host.removeEventListener('dragover', over);
      host.removeEventListener('dragleave', leave);
      host.removeEventListener('drop', drop);
    };
  }, []);

  return (
    <Fragment>
      {children}
      <button
        ref={buttonRef}
        type="button"
        className="ped-image-btn"
        data-editor-ui
        disabled={busy}
        onClick={(event) => {
          // カードはリンクなので、ボタンを押してもページを移動させない
          event.preventDefault();
          event.stopPropagation();
          inputRef.current?.click();
        }}
      >
        {busy ? '保存しています…' : '画像を差し替え'}
      </button>
      {/* リンクの中に置くとクリックがリンクに伝わるので、ファイル選択は body 直下に置く */}
      {createPortal(
        <input
          ref={inputRef}
          type="file"
          accept="image/png,image/jpeg,image/webp,image/gif"
          hidden
          data-editor-ui
          onChange={(event) => {
            replace(event.target.files?.[0]);
            event.target.value = '';
          }}
        />,
        document.body,
      )}
    </Fragment>
  );
}
```

```jsx
// 1項目の入力欄。外をクリックするか Enter で保存する
function Field({ label, code, type, value, onSave, focus }) {
  const ref = useRef(null);
  const initial = type === 'list' ? (value ?? []).join('\n') : type === 'tags' ? (value ?? []).join(', ') : value ?? '';
  const [text, setText] = useState(initial);
  useEffect(() => setText(initial), [initial]);
  useEffect(() => {
    if (!focus) return;
    ref.current?.focus();
    ref.current?.scrollIntoView({ block: 'center' });
  }, [focus]);

  const commit = () => {
    if (text === initial) return;
    if (type === 'list') onSave(text.split('\n').map((line) => line.replace(/^\s*-\s+/, '').trim()).filter(Boolean));
    else if (type === 'tags') onSave(text.split(',').map((item) => item.trim()).filter(Boolean));
    else onSave(text.trim());
  };

  let input;
  if (type === 'fit') {
    input = (
      <select ref={ref} value={value ?? ''} onChange={(event) => onSave(event.target.value)}>
        <option value="">枠いっぱい(はみ出す部分は切る)</option>
        <option value="contain">全体を表示</option>
      </select>
    );
  } else if (type === 'list') {
    input = <textarea ref={ref} rows={4} value={text} onChange={(event) => setText(event.target.value)} onBlur={commit} />;
  } else {
    input = (
      <input
        ref={ref}
        value={text}
        onChange={(event) => setText(event.target.value)}
        onBlur={commit}
        onKeyDown={(event) => {
          if (event.key === 'Enter' && !event.nativeEvent.isComposing && event.keyCode !== 229) {
            event.preventDefault();
            commit();
          }
        }}
      />
    );
  }
  return (
    <label className="pep-field">
      <span>{label}{code && <code>{code}</code>}</span>
      {input}
    </label>
  );
}

const fieldType = (fields, field) => (field === 'tags' ? 'tags' : fields[field] === 'list' ? 'list' : fields[field] === 'fit' ? 'fit' : 'text');

function ItemFields({ item, fields, labels, focusField }) {
  return Object.keys(fields).map((field) => (
    <Field
      key={field}
      label={labels[field]}
      type={fieldType(fields, field)}
      value={item[field]}
      focus={focusField === field}
      onSave={(value) => savePatch(item.source, { field }, value)}
    />
  ));
}

function saveCopy(key, value) {
  if (placeholders(value) !== placeholders(COPY[key])) {
    const ok = window.confirm(`自動で入る部分(${placeholders(COPY[key]) || 'なし'})が変わっています。\nこのまま保存しますか?`);
    if (!ok) return;
  }
  savePatch(COPY_FILE, { key }, value);
}

function CopyGroup({ title, prefix, open }) {
  const keys = Object.keys(COPY).filter((key) => key.startsWith(prefix));
  return (
    <details open={open}>
      <summary>{title}</summary>
      {keys.map((key) => (
        <Field key={key} label="" code={key} type="text" value={COPY[key]} onSave={(value) => saveCopy(key, value)} />
      ))}
    </details>
  );
}

function SettingsPanel({ route }) {
  const { focusField } = useEditorState();
  const work = route.startsWith('/works/') ? workById(route.split('/').pop()) : null;
  return (
    <div>
      {work && (
        <details open>
          <summary>この作品「{work.name}」</summary>
          <ItemFields item={work} fields={WORK_FIELDS} labels={WORK_LABELS} focusField={focusField} />
        </details>

      {route === '/downloads' && DOWNLOADS.map((file) => (

          <summary>ダウンロード「{file.name}」</summary>
          <ItemFields item={file} fields={DOWNLOAD_FIELDS} labels={DOWNLOAD_LABELS} focusField={null} />
        </details>
      ))}
      <CopyGroup title="このページの文言" prefix={prefixFor(route)} open={!work} />
      <CopyGroup title="共通の部品の文言" prefix="common." />
      <CopyGroup title="ナビ・フッター・タブのタイトル" prefix="layout." />
      <details>
        <summary>プロフィールの基本情報</summary>
        {Object.keys(PROFILE_LABELS).map((field) => (
          <Field
            key={field}
            label={PROFILE_LABELS[field]}
            type="text"
            value={SITE.PROFILE[field]}
            onSave={(value) => savePatch(SITE_FILE, { path: `PROFILE.${field}` }, value)}
          />
        ))}
        <Field
          label="「いま取り組んでいること」の時点(2026-10 の形)"
          type="text"
          value={SITE.NOW_UPDATED}
          onSave={(value) => savePatch(SITE_FILE, { path: 'NOW_UPDATED' }, value)}

      </details>
    </div>

}

function ChangesPanel() {
  const { drafts } = useEditorState();
  if (!drafts.length) return <p className="pep-muted">下書きはまだありません。</p>;
  return (
    <ul className="pep-changes">
      {drafts.map((draft) => (
        <li key={draft.file}>
          <div>
            <strong>{draft.label}</strong>
            <code>{draft.file}</code>
            {draft.images.length > 0 && <span className="pep-muted">画像を差し替えています</span>}
            {draft.conflict && (
              <span className="pep-warn">下書きを作り始めた後に、元のファイルが更新されています。公開の前に Claude が確認します。</span>
            )}
          </div>
          <button
            type="button"
            onClick={() => window.confirm(`「${draft.label}」の下書きを取り消します。よろしいですか?`) && discard(draft.file)}
          >
            取り消す
          </button>
        </li>
      ))}
    </ul>
  );
}

function NotesPanel() {
  const { notes } = useEditorState();
  const [text, setText] = useState(notes);
  const [saved, setSaved] = useState(false);
  useEffect(() => setText(notes), [notes]);
  return (
    <div>
      <p className="pep-muted">公開を頼むときに Claude が読みます。「ここは長いかも」「この表現でいいか見てほしい」など、自由に書いてください。</p>
      <textarea
        className="pep-notes"
        rows={8}
        value={text}
        onChange={(event) => {
          setText(event.target.value);
          setSaved(false);
        }}
        onBlur={async () => {
          if (text !== notes && (await saveNotes(text))) setSaved(true);
        }}
      />
      <p className="pep-muted">{saved ? '保存しました' : '入力欄の外をクリックすると保存されます'}</p>
    </div>
  );
}

const TITLES = { changes: '変更した下書き', settings: 'このページの設定', notes: 'Claudeへのメモ' };

        {panel === 'settings' && <SettingsPanel route={route} />}
        {panel === 'notes' && <NotesPanel />}
      </div>
    </aside>
  );
}
```

```jsx
// 画面下の編集バーと、そこから開くパネル
import { Fragment, useEffect } from 'react';
import { createPortal } from 'react-dom';
import { useClickCapture } from './Editable.jsx';
import { Panel } from './Panels.jsx';
import { closePanel, discardAll, openPanel, refresh, setMode, useEditorState } from './store.js';

export function EditorRoot({ route }) {
  const state = useEditorState();
  useClickCapture();

  useEffect(() => {
    refresh();
  }, []);

  useEffect(() => {
    document.documentElement.dataset.editorMode = state.mode;
  }, [state.mode]);

  const confirmDiscardAll = () => {
    if (window.confirm('すべての下書きを取り消します。元に戻せません。よろしいですか?')) discardAll();
  };
  const toggle = (name) => (state.panel === name ? closePanel() : openPanel(name));

  return createPortal(
    <Fragment>
      <Panel route={route} />
      <div className="portfolio-editor-bar" data-editor-ui role="toolbar" aria-label="編集モード">
        <span className="peb-title">編集モード</span>
        <div className="peb-switch" role="group" aria-label="表示の切り替え">
          <button type="button" aria-pressed={state.mode === 'edit'} onClick={() => setMode('edit')}>編集</button>
          <button type="button" aria-pressed={state.mode === 'view'} onClick={() => setMode('view')}>閲覧</button>
        </div>
        <button type="button" className="peb-count" aria-pressed={state.panel === 'changes'} onClick={() => toggle('changes')}>
          変更 {state.drafts.length}件
        </button>
        <button type="button" aria-pressed={state.panel === 'settings'} onClick={() => toggle('settings')}>
          このページの設定
        </button>
        <button type="button" aria-pressed={state.panel === 'notes'} onClick={() => toggle('notes')}>
          Claudeへのメモ{state.notes.trim() ? ' ●' : ''}
        </button>
        <button type="button" className="peb-danger" onClick={confirmDiscardAll} disabled={!state.drafts.length}>
          すべて取り消す
        </button>
        {state.error && <p className="peb-error" role="alert">{state.error}</p>}
      </div>
    </Fragment>,
    document.body,
  );
}
```

`npx vite --mode edit --port 5190 --strictPort` を起動して次を確かめ、最後に「すべて取り消す」で下書きを消してサーバーを止める。

1. 作品詳細(`#/works/gamebar-toki`)で「このページの設定」を開くと、作品の全項目(作品名〜画像の表示)が今の値で並ぶ。

### Task 13: 下書きを本物へ反映する editor-apply

**Files:**
- Create: `scripts/editor-apply.mjs`

**Interfaces:**
- Consumes: `sha256` `DraftStore`(Task 8、テストの準備にも使う)、`parseFrontmatter`(Task 1)
- Produces:
  - `lineDiff(before, after) → string[]`(`- 消えた行` / `+ 足された行`)
  - `planApply(root) → { items: [{ rel, kind: 'text'|'image', status: 'changed'|'new'|'same'|'conflict', diff?, reason? }], conflicts, notes }`(何も変えない)

  - CLI: `npm run editor:apply -- --dry-run`(衝突があれば終了コード 2)、`npm run editor:apply`

```js
import { test } from 'node:test';
import assert from 'node:assert/strict';
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';
import { applyDrafts, lineDiff, planApply } from '../scripts/editor-apply.mjs';
import { DraftStore } from '../scripts/editor/drafts.mjs';
import { applyPatch } from '../src/editor/patch.js';

const SAMPLE = '---\nname: サンプル\nimage: /images/projects/sample.png\nsummary: 紹介です。\n---\n\n## 概要\n\n本文。\n';
const PNG = Buffer.concat([Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]), Buffer.alloc(16)]);
const NOW = new Date(2026, 10, 9, 15, 30, 12);
const WORK = 'src/content/works/sample.md';

function setup() {
  const root = fs.mkdtempSync(path.join(os.tmpdir(), 'apply-'));
  fs.mkdirSync(path.join(root, 'src/content/works'), { recursive: true });
  fs.mkdirSync(path.join(root, 'public/images/projects'), { recursive: true });
  fs.writeFileSync(path.join(root, WORK), SAMPLE);
  fs.writeFileSync(path.join(root, 'public/images/projects/sample.png'), PNG);
  fs.writeFileSync(path.join(root, 'src/content/copy.json'), '{\n  "a": "b"\n}\n');

  const store = new DraftStore(root);
  store.write(WORK, applyPatch(WORK, store.read(WORK), { field: 'summary' }, '新しい紹介です。'));
  const image = store.saveImage(WORK, PNG, { now: NOW });
  store.write(WORK, applyPatch(WORK, store.read(WORK), { field: 'image' }, image, { allowImage: true }));
  store.write('src/content/copy.json', '{\n  "a": "c"\n}\n');
  store.setNotes('ここ長いかも');
  return { root, store, image };
}
const read = (root, rel) => fs.readFileSync(path.join(root, rel), 'utf8');

test('lineDiff は消えた行と足された行を返す', () => {
  assert.deepEqual(lineDiff('a\nb\nc', 'a\nB\nc\nd'), ['- b', '+ B', '+ d']);
  assert.deepEqual(lineDiff('same', 'same'), []);
});

test('確認(dry-run)は何も変えない', () => {
  const { root, image } = setup();
  const plan = planApply(root);
  assert.deepEqual(plan.conflicts, []);
  assert.deepEqual(
    plan.items.map(({ rel, kind, status }) => [rel, kind, status]),
    [
      [`public${image}`, 'image', 'new'],
      ['src/content/copy.json', 'text', 'changed'],
      [WORK, 'text', 'changed'],
    ],
  );
  assert.ok(plan.items.find((item) => item.rel === WORK).diff.includes('+ summary: 新しい紹介です。'));
  assert.equal(plan.notes, 'ここ長いかも');
  assert.equal(read(root, WORK), SAMPLE);
});

test('反映: 本物を書き換え、画像を置き、下書きとメモを控えに移す', () => {
  const { root, store, image } = setup();
  const result = applyDrafts(root, { now: NOW });
  assert.deepEqual(result.applied.sort(), [`public${image}`, 'src/content/copy.json', WORK]);
  assert.match(read(root, WORK), /summary: 新しい紹介です。/);
  assert.ok(fs.existsSync(path.join(root, 'public', image)));
  assert.equal(result.historyDir, path.join(root, '.editor/history/20261109-153012'));
  assert.ok(fs.existsSync(path.join(result.historyDir, 'drafts', WORK)));
  assert.equal(fs.readFileSync(path.join(result.historyDir, 'notes.md'), 'utf8'), 'ここ長いかも');
  assert.deepEqual(store.list(), []);
  assert.equal(store.getNotes(), '');
  assert.deepEqual(result.unusedImages, ['/images/projects/sample.png']);
});

test('ほかから使われている画像は、使われなくなった候補にしない', () => {
  const { root } = setup();
  fs.writeFileSync(path.join(root, 'src/content/works/other.md'), '---\nname: 別\nimage: /images/projects/sample.png\n---\n');
  assert.deepEqual(applyDrafts(root, { now: NOW }).unusedImages, []);
});

test('衝突があれば何も反映しない', () => {
  const { root } = setup();
  fs.writeFileSync(path.join(root, WORK), `${SAMPLE}追記\n`);
  const plan = planApply(root);
  assert.deepEqual(plan.conflicts.map((item) => item.rel), [WORK]);
  assert.throws(() => applyDrafts(root, { now: NOW }), /衝突/);
  assert.equal(read(root, 'src/content/copy.json'), '{\n  "a": "b"\n}\n');
  assert.ok(new DraftStore(root).list().length > 0);
```

```js
#!/usr/bin/env node
// ============================================================
// 編集モードの下書きを本物へ反映する(公開の手順で Claude が使う。docs/editor.md)。
//
//   npm run editor:apply -- --dry-run   差分・衝突・メモを表示するだけ(衝突があれば終了コード 2)
//   npm run editor:apply                反映して、下書きとメモを .editor/history/<日時>/ へ移す
//
// 下書きを作り始めた後に本物が更新されていたら(衝突)、何も反映せずに止まる。
// 差し替えで使われなくなった画像は候補として表示するだけで、消さない。
// ============================================================
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';
import { parseFrontmatter } from '../src/lib/markdown-parse.js';
import { sha256 } from './editor/drafts.mjs';

const ROOT = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');

const paths = (root) => {
  const dir = path.join(root, '.editor');
  return {
    dir,
    drafts: path.join(dir, 'drafts'),
    manifest: path.join(dir, 'manifest.json'),
    notes: path.join(dir, 'notes.md'),
  };
};

function walk(dir, base = dir) {
  if (!fs.existsSync(dir)) return [];
  return fs
    .readdirSync(dir, { withFileTypes: true })
    .flatMap((entry) => {
      const full = path.join(dir, entry.name);
      return entry.isDirectory() ? walk(full, base) : [path.relative(base, full).split(path.sep).join('/')];
    })
    .sort();
}

export function lineDiff(before, after) {
  const a = before.split('\n');
  const b = after.split('\n');
  const lcs = Array.from({ length: a.length + 1 }, () => new Array(b.length + 1).fill(0));
  for (let i = a.length - 1; i >= 0; i--) {
    for (let j = b.length - 1; j >= 0; j--) {
      lcs[i][j] = a[i] === b[j] ? lcs[i + 1][j + 1] + 1 : Math.max(lcs[i + 1][j], lcs[i][j + 1]);
    }
  }
  const out = [];
  let i = 0;
  let j = 0;
  while (i < a.length && j < b.length) {
    if (a[i] === b[j]) {
      i++;
      j++;
    } else if (lcs[i + 1][j] >= lcs[i][j + 1]) {
      out.push(`- ${a[i++]}`);
    } else {
      out.push(`+ ${b[j++]}`);
    }
  }
  while (i < a.length) out.push(`- ${a[i++]}`);
  while (j < b.length) out.push(`+ ${b[j++]}`);
  return out;
}

function readManifest(file) {
  try {
    return JSON.parse(fs.readFileSync(file, 'utf8'));
  } catch {
    return {};
  }
}

export function planApply(root = ROOT) {
  const p = paths(root);
  const manifest = readManifest(p.manifest);
  const items = walk(p.drafts).map((rel) => {
    const draft = fs.readFileSync(path.join(p.drafts, rel));
    const target = path.join(root, rel);
    if (rel.startsWith('public/images/projects/')) {
      return fs.existsSync(target)
        ? { rel, kind: 'image', status: 'conflict', reason: '同じ名前の画像がすでにあります' }
        : { rel, kind: 'image', status: 'new' };
    }
    if (!fs.existsSync(target)) return { rel, kind: 'text', status: 'conflict', reason: '元のファイルがなくなっています' };
    const original = fs.readFileSync(target);
    if (original.equals(draft)) return { rel, kind: 'text', status: 'same' };
    const diff = lineDiff(original.toString('utf8'), draft.toString('utf8'));
    return manifest[rel]?.base === sha256(original)
      ? { rel, kind: 'text', status: 'changed', diff }
      : { rel, kind: 'text', status: 'conflict', diff, reason: '下書きを作り始めた後に、元のファイルが更新されています' };
  });
  const notes = fs.existsSync(p.notes) ? fs.readFileSync(p.notes, 'utf8') : '';
  return { items, conflicts: items.filter((item) => item.status === 'conflict'), notes };
}

// src/ のどこかでその画像のパスが使われているか
function isReferenced(root, image) {
  return walk(path.join(root, 'src')).some((rel) => fs.readFileSync(path.join(root, 'src', rel), 'utf8').includes(image));
}

const stamp = (d) =>
  `${d.getFullYear()}${String(d.getMonth() + 1).padStart(2, '0')}${String(d.getDate()).padStart(2, '0')}-` +
  `${String(d.getHours()).padStart(2, '0')}${String(d.getMinutes()).padStart(2, '0')}${String(d.getSeconds()).padStart(2, '0')}`;

export function applyDrafts(root = ROOT, { now = new Date() } = {}) {
  const p = paths(root);
  const plan = planApply(root);
  if (plan.conflicts.length) {
    throw new Error(`衝突があるため反映しませんでした: ${plan.conflicts.map((item) => item.rel).join(', ')}`);
  }
  if (!plan.items.length) return { applied: [], historyDir: null, unusedImages: [] };

  const applied = [];
  const replacedImages = [];
  for (const item of plan.items) {
    if (item.status === 'same') continue;
    const source = path.join(p.drafts, item.rel);
    const target = path.join(root, item.rel);
    if (item.rel.endsWith('.md')) {
      const before = parseFrontmatter(fs.readFileSync(target, 'utf8')).data.image;
      const after = parseFrontmatter(fs.readFileSync(source, 'utf8')).data.image;
      if (before && before !== after) replacedImages.push(before);
    }
    fs.mkdirSync(path.dirname(target), { recursive: true });
    fs.copyFileSync(source, target);
    applied.push(item.rel);
  }

  const historyDir = path.join(p.dir, 'history', stamp(now));
  fs.mkdirSync(historyDir, { recursive: true });
  fs.renameSync(p.drafts, path.join(historyDir, 'drafts'));
  if (fs.existsSync(p.manifest)) fs.renameSync(p.manifest, path.join(historyDir, 'manifest.json'));
  if (fs.existsSync(p.notes)) {
    fs.copyFileSync(p.notes, path.join(historyDir, 'notes.md'));
    fs.writeFileSync(p.notes, '');
  }
  fs.writeFileSync(p.manifest, '{}\n');

  const unusedImages = replacedImages.filter((image) => !isReferenced(root, image));
  return { applied, historyDir, unusedImages };
}

function printPlan(plan) {
  if (!plan.items.length) {
    console.log('下書きはありません。');
    return;
  }
  for (const item of plan.items) {
    const label = { changed: '変更', new: '新しい画像', same: '変更なし', conflict: '衝突' }[item.status];
    console.log(`\n[${label}] ${item.rel}${item.reason ? `(${item.reason})` : ''}`);
    for (const line of item.diff ?? []) console.log(`    ${line}`);
  }
  console.log(plan.notes.trim() ? `\n[Claudeへのメモ]\n${plan.notes.trim()}` : '\n[Claudeへのメモ] なし');
}

if (import.meta.url === pathToFileURL(process.argv[1] ?? '').href) {
  const plan = planApply();
  printPlan(plan);
  if (process.argv.includes('--dry-run')) {
    process.exit(plan.conflicts.length ? 2 : 0);
  }
  try {
    const result = applyDrafts();
    console.log(`\n反映しました(${result.applied.length}件)。下書きの控え: ${result.historyDir ? path.relative(ROOT, result.historyDir) : 'なし'}`);
    if (result.unusedImages.length) {
      console.log('使われなくなった画像(削除はしていません):');
      for (const image of result.unusedImages) console.log(`  public${image}`);
    }
  } catch (err) {
    console.error(`\n${err.message}`);
    process.exit(2);
  }
}
```
